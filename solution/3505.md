You are given an integer array nums and two integers, x and k. You can perform the following operation any number of times (including zero):

Increase or decrease any element of nums by 1.
Return the minimum number of operations needed to have at least k non-overlapping subarrays of size exactly x in nums, where all elements within each subarray are equal.

 

Example 1:

Input: nums = [5,-2,1,3,7,3,6,4,-1], x = 3, k = 2

Output: 8

Explanation:

Use 3 operations to add 3 to nums[1] and use 2 operations to subtract 2 from nums[3]. The resulting array is [5, 1, 1, 1, 7, 3, 6, 4, -1].
Use 1 operation to add 1 to nums[5] and use 2 operations to subtract 2 from nums[6]. The resulting array is [5, 1, 1, 1, 7, 4, 4, 4, -1].
Now, all elements within each subarray [1, 1, 1] (from indices 1 to 3) and [4, 4, 4] (from indices 5 to 7) are equal. Since 8 total operations were used, 8 is the output.
Example 2:

Input: nums = [9,-2,-2,-2,1,5], x = 2, k = 2

Output: 3

Explanation:

Use 3 operations to subtract 3 from nums[4]. The resulting array is [9, -2, -2, -2, -2, 5].
Now, all elements within each subarray [-2, -2] (from indices 1 to 2) and [-2, -2] (from indices 3 to 4) are equal. Since 3 operations were used, 3 is the output.
 

Constraints:

2 <= nums.length <= 105
-106 <= nums[i] <= 106
2 <= x <= nums.length
1 <= k <= 15
2 <= k * x <= nums.length

default :
import java.util.*;

class Solution {

    static class Fenwick {
        long[] bit;
        int n;

        Fenwick(int n) {
            this.n = n;
            bit = new long[n + 2];
        }

        void add(int idx, long delta) {
            while (idx <= n) {
                bit[idx] += delta;
                idx += idx & -idx;
            }
        }

        long query(int idx) {
            long res = 0;
            while (idx > 0) {
                res += bit[idx];
                idx -= idx & -idx;
            }
            return res;
        }

        long query(int l, int r) {
            return query(r) - query(l - 1);
        }

        int kth(long k) {
            int pos = 0;
            int bitMask = Integer.highestOneBit(n);

            for (int d = bitMask; d != 0; d >>= 1) {
                int next = pos + d;
                if (next <= n && bit[next] < k) {
                    k -= bit[next];
                    pos = next;
                }
            }
            return pos + 1;
        }
    }

    public long minOperations(int[] nums, int x, int k) {
        int n = nums.length;

        long[] cost = buildCosts(nums, x);

        int m = cost.length;

        long INF = Long.MAX_VALUE / 4;

        long[][] dp = new long[k + 1][m + 1];

        for (int i = 0; i <= k; i++) {
            Arrays.fill(dp[i], INF);
        }

        dp[0][0] = 0;

        for (int i = 0; i < m; i++) {

            for (int t = 0; t <= k; t++) {

                if (dp[t][i] == INF) continue;

                dp[t][i + 1] =
                        Math.min(dp[t][i + 1], dp[t][i]);

                if (t < k) {

                    int next = Math.min(m, i + x);

                    dp[t + 1][next] =
                            Math.min(
                                    dp[t + 1][next],
                                    dp[t][i] + cost[i]
                            );
                }
            }
        }

        long ans = INF;

        for (int i = 0; i <= m; i++) {
            ans = Math.min(ans, dp[k][i]);
        }

        return ans;
    }

    private long[] buildCosts(int[] nums, int x) {

        int n = nums.length;

        int[] vals = nums.clone();
        Arrays.sort(vals);

        int ptr = 0;
        for (int v : vals) {
            if (ptr == 0 || vals[ptr - 1] != v) {
                vals[ptr++] = v;
            }
        }

        int[] uniq = Arrays.copyOf(vals, ptr);

        Fenwick countBIT = new Fenwick(ptr);
        Fenwick sumBIT = new Fenwick(ptr);

        int[] idx = new int[n];

        for (int i = 0; i < n; i++) {
            idx[i] = Arrays.binarySearch(uniq, nums[i]) + 1;
        }

        for (int i = 0; i < x; i++) {
            countBIT.add(idx[i], 1);
            sumBIT.add(idx[i], nums[i]);
        }

        long[] res = new long[n - x + 1];

        for (int start = 0; start <= n - x; start++) {

            long need = (x + 1L) / 2;

            int medianPos = countBIT.kth(need);
            long median = uniq[medianPos - 1];

            long leftCnt = countBIT.query(medianPos);
            long leftSum = sumBIT.query(medianPos);

            long totalCnt = x;
            long totalSum = sumBIT.query(ptr);

            long rightCnt = totalCnt - leftCnt;
            long rightSum = totalSum - leftSum;

            long cost =
                    median * leftCnt - leftSum
                    + rightSum - median * rightCnt;

            res[start] = cost;

            if (start == n - x) break;

            countBIT.add(idx[start], -1);
            sumBIT.add(idx[start], -nums[start]);

            countBIT.add(idx[start + x], 1);
            sumBIT.add(idx[start + x], nums[start + x]);
        }

        return res;
    }
}
